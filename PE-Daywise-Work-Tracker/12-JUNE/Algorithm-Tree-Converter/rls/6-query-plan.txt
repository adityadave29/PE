SELECT
    o.o_orderpriority,
    COUNT(*)
FROM orders o

/* EXISTS(lineitem) became Hash Semi Join */
SEMI JOIN lineitem l
    ON l.l_orderkey = o.o_orderkey

WHERE
    o.o_orderdate >= DATE '1994-01-01'
    AND o.o_orderdate < DATE '1994-04-01'

    /* RLS policy became hashed SubPlan */
    AND o.o_custkey IN (
        SELECT c.c_custkey
        FROM customer c
        WHERE c.c_acctbal > 5000
    )

    /* lineitem filter */
    AND l.l_commitdate < l.l_receiptdate

GROUP BY o.o_orderpriority
ORDER BY o.o_orderpriority;
